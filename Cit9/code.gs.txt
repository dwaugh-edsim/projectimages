function doGet(e) {
  const action = e.parameter.action;
  const studentId = e.parameter.studentId;
  const pin = e.parameter.pin;
  const missionTitle = e.parameter.missionTitle;

  // Handle Login Action
  if (action === 'LOGIN') {
    const correctPin = PIN_LIST[studentId];
    const isSuccess = (correctPin && correctPin === (pin || '').toUpperCase());
    
    return ContentService.createTextOutput(JSON.stringify({ success: isSuccess }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Handle Data Retrieval
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const rows = sheet.getDataRange().getValues();
  
  // Find the most recent entry for this student and mission
  let content = null;
  for (let i = rows.length - 1; i >= 1; i--) {
    if (rows[i][0] === studentId && rows[i][1] === missionTitle) {
      content = rows[i][2]; // Assumes Col A: studentId, Col B: missionTitle, Col C: reflection (JSON)
      break;
    }
  }

  return ContentService.createTextOutput(JSON.stringify({ content: content }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const payload = JSON.parse(e.postData.contents);
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  
  // Append new row: StudentId, MissionTitle, ReflectionJSON, Timestamp
  sheet.appendRow([
    payload.studentId,
    payload.missionTitle,
    payload.reflection,
    payload.timestamp
  ]);

  return ContentService.createTextOutput(JSON.stringify({ success: true }))
    .setMimeType(ContentService.MimeType.JSON);
}
